// bit_swap 函数保持不变
static uint16_t bit_swap(uint16_t src)
{
    uint16_t dst = 0, i = 0;
    for(i = 0; i < 16; i++)
        dst |= ((i < 8) ? ((src & (1 << i)) << (15 - i * 2)) : ((src & (1 << i)) >> (2 * i - 15)));
    return dst;
}

// write_icap_data 函数保持不变，但其内部调用的mdio_word_write已适配
static int write_icap_data(uint32_t *pdata)
{
    uint32_t data;
    uint32_t data_out;
    uint16_t data_hi, data_lo;
    uint16_t data_hi_swap, data_lo_swap;

    data = *pdata;
            
    /* swap low and high */
    data_hi = (uint16_t)(data >> 16) & 0xffff;
    data_lo = (uint16_t)data & 0xffff;

    data_hi_swap =/* bit_swap*/(data_hi);
    data_lo_swap =/* bit_swap*/(data_lo);

    data_out = ((uint32_t)data_hi_swap) << 16 | ((uint32_t)data_lo_swap);
    mdio_word_write(ICAP_WR_REG, data_out); // 调用适配后的 mdio_word_write

    return 0;
}

// fpga_icap_config 函数保持不变，但其内部调用的mdio_word_write已适配
int fpga_icap_config(char *file)
{
    int      fd = -1;
    int      i;
    uint32_t *pw = NULL;
    uint32_t file_size = 0, progress;
    unsigned char* buffer = NULL;

    /* read bin file into buffer */
    if((fd = open(file,O_RDONLY)) == -1){
        fprintf(stderr, "open file %s failed!\n",file);
        return -1;
    }
    file_size = lseek(fd, 0, SEEK_END);
    fprintf(stdout,"%s : bin file size %dbytes\n",__func__,file_size);
    if((file_size % 2) != 0){
        printf("ERROR! The file %s size is even number\n", file);
        close(fd); // Add close(fd) here for error path
        return -1;
    }
    lseek(fd, 0, SEEK_SET);
    buffer = (unsigned char*)calloc(file_size, sizeof(unsigned char));
    if(!buffer){
        close(fd);
        fprintf(stderr,"%s : calloc failed!\n",__func__);
        return -1;
    }
    if(read(fd, buffer, file_size) != file_size){
        free(buffer);
        close(fd);
        fprintf(stderr,"%s : read file to buffer error!\n",__func__);
        return -1;
    }
    close(fd);

    // enable icap configuration
    mdio_word_write(ICAP_CTL_REG, 0x2); // 调用适配后的 mdio_word_write

    /* now program data into icap */
    pw = (uint32_t *)buffer;
    {
        printf("Start Programming...\n");
        for (i = 0; i < file_size/4; i++){
            write_icap_data(pw++); // 调用适配后的 write_icap_data
            if (i % 100 == 0) {
                progress = (i*400)/file_size; // Note: original code might be (i * 100 * 4 / file_size) for percentage, 400 is odd.
                printf("[%d%%]\r", progress);
                fflush(stdout);
            }
        }
        printf("loop i = %d\n", i);
        printf("End Programming...\n");
    }
    free(buffer); // Release allocated memory
    return 0;
}

// preparebinfile 函数保持不变
static int preparebinfile(char *name, int *fileSize)
{
    int  iRet=-1;
    FILE *hBitFileIn, *hBitFileOut;
    char buffer[SPIFLASH_PAGESIZE];
    int  pBegin, pEnd, fSize, nPages;
    int  rdCount;
    int  padCnt = 0;
    int  flag_first_page = 1;
    char ch = 255;//FF
    
    if( (hBitFileIn=fopen(name,"rb"))==NULL ){
        printf("can't open file %s for read\n", name);
        return iRet;
    }
    if( (hBitFileOut=fopen("./FPGA.bin","wb"))==NULL ){
        printf("can't open file FPGA.bin for write\n");
        fclose(hBitFileIn); // Close input file if output fails
        return iRet;
    }
    
    // calculate file size, align to page size
    fseek(hBitFileIn, 0, SEEK_SET);
    pBegin = ftell(hBitFileIn);
    fseek(hBitFileIn, 0, SEEK_END);
    pEnd = ftell(hBitFileIn);
    fSize = pEnd - pBegin;
    printf("origin file size=%d\n", fSize);
    if ( fSize < SPIFLASH_PAGESIZE ){
        printf("input file is too small!\n");
        fclose(hBitFileIn);
        fclose(hBitFileOut);
        return -1;
    }
    if ( fSize % SPIFLASH_PAGESIZE ){
        padCnt = SPIFLASH_PAGESIZE - (fSize % SPIFLASH_PAGESIZE); 
        nPages = fSize / SPIFLASH_PAGESIZE;
        fSize  = (nPages + 1) * SPIFLASH_PAGESIZE;
        printf("nPages=%d, fSize=%d, padCnt=%d\n", nPages, fSize, padCnt); 
    }
    *fileSize = fSize;
    
    // begin at 0
    fseek(hBitFileIn, 0, SEEK_SET);
    while ( (rdCount = fread(buffer, 1, SPIFLASH_PAGESIZE, hBitFileIn)) > 0 ){
        // replace AA 99 55 66
        if ( flag_first_page ){
            // These bytes are specific to the original bitstream format
            // Ensure this modification is still desired for the new context
           // buffer[48] = 0xFF;
           // buffer[49] = 0xFF;
           // buffer[50] = 0xFF;
           // buffer[51] = 0xFF;
            flag_first_page = 0;
        }
        fwrite(buffer, rdCount, 1, hBitFileOut);
    }
    // pad to PAGESIZE
    while ( padCnt > 0 ){
        fwrite(&ch, 1, 1, hBitFileOut);
        padCnt--;
    }
    // close file
    fclose(hBitFileIn);
    fclose(hBitFileOut);
    
    iRet=0;

// _Exit: // Label not needed if no goto Jump here
    return iRet;
}

// reconfig 函数保持不变，但其内部调用的mdio_word_write已适配
static void reconfig()

// ==============================================================================
// 程序B的主函数 (修改和添加了文件读取和FPGA配置逻辑)
// ==============================================================================
int main(int argc, char *argv[])
{
    // 将所有局部变量声明移到函数的开头
    char *name;
    int opt;
    int ret = 0; // 声明并初始化
    int addr = 0; // 声明并初始化
    int value = 0; // 声明并初始化

    char *device = NULL; // 声明并初始化
    // struct mqnic *dev; // Already declared for global_mqnic_dev, this line can be removed
    struct mqnic_reg_block *rb;

    int interface = 0; // 声明并初始化
    int port = 0; // 声明并初始化
    int verbose = 0; // 声明并初始化
    int n;

    // Parameters for FPGA configuration from Program A
    char *bin_file_path = NULL; // 声明并初始化

    // FPGA configuration specific variables
    int filesize;
    time_t start_time;


    name = strrchr(argv[0], '/');
    name = name ? 1+name : argv[0];

    // Added 'f:' for bin file path argument
    while ((opt = getopt(argc, argv, "d:i:P:vf:h?")) != EOF)
    {
        switch (opt)
        {
        case 'd':
            device = optarg;
            break;
        case 'i':
            interface = atoi(optarg);
            break;
        case 'P':
            port = atoi(optarg);
            break;
        case 'v':
            verbose++;
            break;
        case 'f': // New option for bin file path
            bin_file_path = optarg;
            break;
        case 'h':
        case '?':
            fprintf(stderr, "Usage: %s -d <device> [-i <interface>] [-P <port>] [-v] [-f <bin_file_path>]\n", name);
            return 0;
        default:
            fprintf(stderr, "Unknown option: %c\n", opt);
            return -1;
        }
    }
    
    if (!device)
    {
        fprintf(stderr, "Device not specified\n");
        return -1;
    }

    // Open mqnic device
    global_mqnic_dev = mqnic_open(device); // Assign to global variable

    if (!global_mqnic_dev)
    {
        fprintf(stderr, "Failed to open device\n");
        return -1;
    }
    
    rb = mqnic_find_reg_block(global_mqnic_dev->rb_list, MQNIC_RB_APP_INFO_TYPE, MQNIC_RB_APP_INFO_VER, 0);
    if (rb) {
        global_mqnic_dev->app_id = mqnic_reg_read32(rb->regs, MQNIC_RB_APP_INFO_REG_ID);

        if (!global_mqnic_dev->app_regs) {
            fprintf(stderr, "Warning: application section present, but application BAR not mapped\n");
        }
    } else {
        fprintf(stderr, "Error: Application info register block not found. Cannot access app_regs.\n");
        mqnic_close(global_mqnic_dev);
        return -1;
    }

    // ==========================================================================
    // 从程序A导入的FPGA配置逻辑
    // ==========================================================================
    if (bin_file_path) {
        printf("\n--- Starting FPGA Configuration ---\n");
        if (preparebinfile(bin_file_path, &filesize) < 0 ){
            printf("Error, Can't prepare bin file %s\n", bin_file_path);
            mqnic_close(global_mqnic_dev);
            return -2;
        }
        start_time = time(NULL);
        if (updatebinfile(filesize) < 0 ){
            printf("Error, Fail to update bin file to flash\n");
            mqnic_close(global_mqnic_dev);
            return -1;
        }
        reconfig();
        printf("Total Flashing time: %ld s\n",time(NULL)-start_time);
        printf("--- FPGA Configuration Complete ---\n\n");
    }

    // ==========================================================================
    // 原始程序B的交互式寄存器读写逻辑
    // ==========================================================================
    while(1){
        printf("1. Read reg  2. Write reg  3. Quit  4. (Optional) Reconfigure FPGA\n");
        scanf("%d", &n);
        if (n == 3) break;
        switch(n){
          case 1: printf("Input the address (hex): \n");
            scanf("%x",&addr);
            printf("Read 0x%x, Value 0x%x\n", addr, mqnic_reg_read32(global_mqnic_dev->app_regs, addr));
            break;
          case 2: printf("Input the address (hex) and write value (hex): \n");
            scanf("%x %x",&addr,&value);
            mqnic_reg_write32(global_mqnic_dev->app_regs, addr, value);
            printf("Written 0x%x to 0x%x\n", value, addr);
            break;
          // case 4: // If you want to add a trigger for reconfig again
          //   reconfig();
          //   printf("FPGA reconfig command sent.\n");
          //   break;
          default: 
            printf("Invalid option.\n");
            break;
        }
    }

    // Close mqnic device
    mqnic_close(global_mqnic_dev);
    return 0;
}
